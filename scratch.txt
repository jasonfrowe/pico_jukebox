// --- DRUM DEFINITIONS ---
const OPL_Patch drum_bd = { .m_ave=0x00, .m_ksl=0x00, .m_atdec=0xF1, .m_susrel=0xCF, .m_wave=0x00, .c_ave=0x00, .c_ksl=0x00, .c_atdec=0xF1, .c_susrel=0xCF, .c_wave=0x00, .feedback=0x00 };

// --- INTERNAL HELPER ---
void write_patch_to_channel(uint8_t ch, const OPL_Patch* p) {
    if (ch > 8) return;
    uint8_t offsets[9] = {0, 1, 2, 8, 9, 10, 16, 17, 18};
    uint8_t off = offsets[ch];

    // Modulator
    opl_write(false, 0x20 + off); opl_write(true, p->m_ave);
    opl_write(false, 0x40 + off); opl_write(true, p->m_ksl);
    opl_write(false, 0x60 + off); opl_write(true, p->m_atdec);
    opl_write(false, 0x80 + off); opl_write(true, p->m_susrel);
    opl_write(false, 0xE0 + off); opl_write(true, p->m_wave);

    // Carrier
    opl_write(false, 0x23 + off); opl_write(true, p->c_ave);
    opl_write(false, 0x43 + off); opl_write(true, p->c_ksl);
    opl_write(false, 0x63 + off); opl_write(true, p->c_atdec);
    opl_write(false, 0x83 + off); opl_write(true, p->c_susrel);
    opl_write(false, 0xE3 + off); opl_write(true, p->c_wave);

    // Feedback
    opl_write(false, 0xC0 + ch);  opl_write(true, p->feedback);
}

// --- PUBLIC API IMPLEMENTATION ---

void load_gm_instrument(uint8_t channel, uint8_t program_number) {
    if (program_number > 127) program_number = 0; // Fallback to Piano
    
    // Check if the patch is "Empty" (all zeros) and fallback to piano if so
    // (Optional safety check if you didn't populate the whole array)
    
    write_patch_to_channel(channel, &gm_bank[program_number]);
}

void load_drum_patch(uint8_t channel, uint8_t drum_note) {
    if (drum_note == 35 || drum_note == 36) write_patch_to_channel(channel, &drum_bd);
    else if (drum_note == 38 || drum_note == 40) write_patch_to_channel(channel, &drum_snare);
    else write_patch_to_channel(channel, &drum_hihat);
}
